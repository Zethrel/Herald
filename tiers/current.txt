# ---- Everyone --------------------------------------------------------------
#
# The lowest-priority block: applies to every spec, whatever it plays. Food and
# the main-stat potion live here.

# Provisional — the season has not started, so the newer feasts are not out yet.
all: food = Harandar Celebration

# Main stat is what the potion carries now, and this one is the main-stat potion
# across the board. Some specs run Potion of Recklessness instead: give those a
# spec line below.
all: potion = Light's Potential


# ---- Flasks, by secondary stat ----------------------------------------------
#
# Modern flasks give a secondary stat, not a primary one, so the flask follows
# the stat priority rather than the class. Each spec picks one of these with its
# own "secondary = ..." line further down.

crit:         flask = Flask of the Shattered Sun; oil = Thalassian Phoenix Oil
haste:        flask = Flask of the Blood Knights; oil = Thalassian Phoenix Oil
mastery:      flask = Flask of the Magisters;      oil = none
versatility:  flask = Flask of Thalassian Resistance; oil = none

# Thalassian Phoenix Oil covers crit and haste, and no mastery or versatility
# oil exists yet. "none" says exactly that: those specs are told there is no oil
# this tier, rather than being shown a blank they have to go and check.


# ---- Per role ---------------------------------------------------------------
#
# Role beats stat: every healer is an intellect user, so a healer line here wins
# for healers while any stat line still covers everyone else.
